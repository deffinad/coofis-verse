import { createSwapy } from "swapy";
import { useEffect, useRef, useState } from "react";
import {
  Box,
  Paper,
  Typography,
  Button,
  MenuItem,
  Select,
} from "@mui/material";
import { Components } from "remoteApp/Components";

const STORAGE_KEY = "swapy_rows";

const DraggableItem = ({ text, onDelete }) => {
  return (
    <div style={{ display: "flex", justifyContent: "space-between" }}>
      <Typography variant="h6">{text}</Typography>
      <Button variant="contained" color="error" size="small" onClick={onDelete}>
        Hapus
      </Button>
    </div>
  );
};

const Swapyswapy = () => {
  const swapy = useRef(null);
  const container = useRef(null);
  const [rows, setRows] = useState(() => {
    const savedRows = localStorage.getItem(STORAGE_KEY);
    return savedRows ? JSON.parse(savedRows) : [];
  });

  const [selectedRow, setSelectedRow] = useState(null); // Menyimpan layout yang dipilih

  useEffect(() => {
    localStorage.setItem(STORAGE_KEY, JSON.stringify(rows));
  }, [rows]);

  const initializeSwapy = () => {
    if (swapy.current?.destroy) {
      swapy.current.destroy();
    }

    if (container.current) {
      swapy.current = createSwapy(container.current);
      swapy.current.onSwap((event) => {
        console.log("swap", event);
      });
    }
  };

  useEffect(() => {
    initializeSwapy();
  }, [rows]);

  // Tambah Row (Layout)
  const addGrid = () => {
    const newRow = {
      idDroppable: Date.now(),
      column: 12,
      name: "Grid",
      components: null,
      props: null,
    };
    setRows((prevRows) => [...prevRows, newRow]);
  };

  // Tambah Kolom di dalam Row yang dipilih
  const addRatings = () => {
    if (selectedRow === null) return;

    setRows((prevRows) =>
      prevRows.map((row) =>
        row.idDroppable === selectedRow
          ? {
              ...row,
              components: [
                ...(row.components || []),
                {
                  idDroppable: Date.now(),
                  column: 12,
                  components: "Ratings",
                  props: null,
                },
              ],
            }
          : row
      )
    );
  };

  // Hapus Layout (Row)
  const removeLayout = () => {
    if (selectedRow === null) return;

    setRows((prevRows) =>
      prevRows.filter((row) => row.idDroppable !== selectedRow)
    );
    setSelectedRow(null);
  };

  console.log(rows);

  return (
    <Box>
      <Button variant="contained" onClick={addGrid}>
        Tambah Grid
      </Button>

      {/* Button di luar layout */}
      <Button
        variant="outlined"
        sx={{ ml: 2 }}
        disabled={selectedRow === null}
        onClick={addRatings}
      >
        Tambah Ratings
      </Button>

      <Button
        variant="contained"
        color="error"
        sx={{ ml: 2 }}
        disabled={selectedRow === null}
        onClick={removeLayout}
      >
        Hapus Layout
      </Button>

      <Box
        ref={container}
        p={2}
        sx={{ border: "", display: "flex", flexDirection: "column", gap: 2 }}
      >
        {rows.map((row, rowIndex) => (
          <Box
            key={row.idDroppable}
            sx={{
              width: "100%",
              border: selectedRow === row.idDroppable ? "2px dashed green" : "2px dashed red",
              cursor: "pointer",
              minHeight: "100px",
              borderRadius: "5px"
            }}
            onClick={() => setSelectedRow(row.idDroppable)}
          >
            
            {/* <Box sx={{ p: 2 }}> */}
              {/* <Typography variant="subtitle1" sx={{ mb: 1 }}>
                Layout {row.idDroppable}
              </Typography> */}

              {/* {(row.components || []).length === 0 ? (
                <Typography
                  variant="body2"
                  sx={{ fontStyle: "italic", color: "gray" }}
                >
                  (Kosong - Tambahkan Komponen)
                </Typography>
              ) : (
                <Box display="flex" gap={2}>
                  {(row.components || []).map((item, itemIndex) => (
                    <Box
                      key={item.idDroppable}
                      data-swapy-slot={`${rowIndex}-${itemIndex}`}
                      sx={{
                        flex: item.column / 12,
                        minWidth: "100px",
                      }}
                    >
                      <Box
                        data-swapy-item={`${rowIndex}-${itemIndex}`}
                        sx={{
                          border: "1px dashed black",
                          p:2
                        }}
                      >
                        <DraggableItem
                          text={`Komponen ${itemIndex + 1}`}
                          onDelete={() => console.log("Hapus Kolom")}
                        />
                      </Box>
                    </Box>
                  ))}
                </Box>
              )} */}
            {/* </Box> */}
          </Box>
        ))}
      </Box>
    </Box>
  );
};

export default Swapyswapy;
